############################################################
# ASYNCHRONOUS FIFO - SDC CONSTRAINTS
############################################################

# ==========================================================
# WRITE CLOCK
# ==========================================================

create_clock \
    -name wclk \
    -period 10.000 \
    -waveform {0.000 5.000} \
    [get_ports wclk]


# ==========================================================
# READ CLOCK
# ==========================================================

create_clock \
    -name rclk \
    -period 14.000 \
    -waveform {0.000 7.000} \
    [get_ports rclk]


# ==========================================================
# ASYNCHRONOUS CLOCK DOMAINS
# ==========================================================

set_clock_groups \
    -asynchronous \
    -group [get_clocks wclk] \
    -group [get_clocks rclk]


# ==========================================================
# ASYNCHRONOUS RESETS
# ==========================================================

set_false_path \
    -from [get_ports wrst_n]
set_false_path \                                                                                                      8,0-1         Topset_false_path \
set_false_path \                                                                                                      8,0-1         Topset_false_path \
     -from [get_ports rrst_n]
